cmake_minimum_required(VERSION 3.20)
project(wgt C)
# -DBACKEND=dawn  : prebuilt Dawn install tree (static libwebgpu_dawn), via -DDawn_DIR=<prefix>/lib*/cmake/Dawn
# -DBACKEND=wgpu  : wgpu-native release zip, via -DWGPU_ROOT=<unzipped dir>
set(BACKEND "wgpu" CACHE STRING "dawn or wgpu")
add_executable(wgt wgt.c)
if(BACKEND STREQUAL "dawn")
  enable_language(CXX)                      # Dawn is C++ inside; the final link needs the C++ runtime
  find_package(Dawn REQUIRED CONFIG)
  find_package(Threads REQUIRED)
  target_link_libraries(wgt PRIVATE dawn::webgpu_dawn)
  set_target_properties(wgt PROPERTIES LINKER_LANGUAGE CXX)
else()
  target_include_directories(wgt PRIVATE ${WGPU_ROOT}/include)
  if(WIN32)
    target_link_libraries(wgt PRIVATE ${WGPU_ROOT}/lib/wgpu_native.dll.lib)
    add_custom_command(TARGET wgt POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy ${WGPU_ROOT}/lib/wgpu_native.dll $<TARGET_FILE_DIR:wgt>)
  elseif(APPLE)
    target_link_libraries(wgt PRIVATE ${WGPU_ROOT}/lib/libwgpu_native.dylib)
    set_target_properties(wgt PROPERTIES BUILD_RPATH ${WGPU_ROOT}/lib)
  else()
    target_link_libraries(wgt PRIVATE ${WGPU_ROOT}/lib/libwgpu_native.so)
    set_target_properties(wgt PROPERTIES BUILD_RPATH ${WGPU_ROOT}/lib)
  endif()
endif()
